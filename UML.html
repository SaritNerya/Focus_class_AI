01 / CLIENT

שכבת ה-Vision והכיול

המרה של פריים וידאו גולמי לתכונות פנים מספריות, ולאחר מכן לתכונות "מכוילות" ביחס לבסיס האישי של התלמידה (Facade + מקור-אמת יחיד לכיול).

classDiagram
    class Landmark {
        <<Protocol>>
        +x: float
        +y: float
    }

    class FaceMeshDetector {
        -_face_mesh
        -_last_raw_landmarks
        +process(frame_rgb) List~Landmark~
        +draw(frame, color)
        +close()
    }

    class FrameFeatures {
        <<dataclass, frozen>>
        +avg_ear: float
        +mar: float
        +pitch: float
        +yaw: float
        +roll: float
        +nose_pos: Point2D
    }

    class FaceFeatureExtractor {
        +extract(landmarks, img_w, img_h) FrameFeatures
    }

    class VisionMath {
        <<module functions>>
        +calculate_ear(landmarks, eye_idx) float
        +calculate_mar(landmarks, mouth_idx) float
        +calculate_head_pose(landmarks, w, h) HeadPose
        +calculate_movement_distance(cur, prev) float
    }

    class CalibrationConfig {
        <<dataclass>>
        +duration_seconds: float = 3.0
        +ear_std_dev_multiplier: float = 2.0
        +ear_threshold_min/max: float
        +angle_smoothing_alpha: float = 0.4
    }

    class Calibrator {
        -_samples: List
        -_config: CalibrationConfig
        +is_calibrated: bool
        +baseline_ear/yaw/roll: float
        +ear_threshold: float
        +remaining_seconds() float
        +add_sample(features, dt_seconds)
        +calibrate_frame(features) CalibratedFeatures

    class CalibratedFeatures {
        <<dataclass, frozen>>
        +avg_ear: float
        +delta_ear: float
        +adjusted_yaw: float

    FaceMeshDetector ..> Landmark : produces
    FaceFeatureExtractor ..> Landmark : reads

    Calibrator o-- CalibrationConfig : configured by
    Calibrator ..> FrameFeatures : consumes
    Calibrator --> CalibratedFeatures : creates
תיקון ארכיטקטוני: בקוד המקורי היה כיול כפול וסותר (ב-main.py וגם בתוך scoring.py). כאן Calibrator הוא מקור האמת היחיד — כל שאר המערכת מקבלת CalibratedFeatures מוכנות.
02 / CLIENT — DESIGN PATTERN: DECORATOR

מנוע הניקוד — שרשרת ה-Decorator

הליבה הארכיטקטונית של המערכת. כל "עונש" הוא מחלקה שעוטפת FocusEvaluator קיים ומוסיפה שכבת התנהגות — הוספת עונש חדש דורשת מחלקה אחת בלבד, ללא שינוי בקוד קיים (Open/Closed).

classDiagram
    class Verdict {
        <<dataclass, frozen>>
        +target_ceiling: float = 100.0
        +reasons: Tuple~str~
        +is_violating: bool
        +is_severe: bool
    }

    class FocusEvaluator {
        <<abstract>>
        +evaluate(features) Verdict*
        +reset()
    }

    class BaseEvaluator {
        "Concrete Component — 100 תמיד"
        +evaluate(features) Verdict
    }

    class PenaltyDecorator {
        <<abstract>>
        #_wrapped: FocusEvaluator
        +evaluate(features) Verdict
        #_evaluate_self(features) Verdict*
        #_reset_self()
    }

    class MovementPenaltyDecorator {
        -_movement_frames: float
        -_prev_nose_pos
        +seconds_moving() float
        #_evaluate_self(features) Verdict
    }

    class EyesClosedPenaltyDecorator {
        -_closed_eyes_frames: int
        #_evaluate_self(features) Verdict
    }

    class TiltPenaltyDecorator {
        -_distracted_frames: int
        #_evaluate_self(features) Verdict
    }

    class YawnPenaltyDecorator {
        -_yawn_frames: int
        #_evaluate_self(features) Verdict
    }

    class ScoringConfig {
        <<dataclass, frozen>>
        +movement_threshold_px +frames_blink_threshold
        +tilt_threshold_deg +mar_threshold
        +severe_*_seconds +slow/normal_recovery_rate
        "כל הספים — Dependency Injection"
    }

    class ScoreEngine {
        -_evaluator: FocusEvaluator
        -_config: ScoringConfig
        +score: float
        -_severe_recovery_active: bool
        +update(features) ScoreResult
        +reset()
    }

    class ScoreResult {
        <<dataclass, frozen>>
        +score: float
        +reasons: Tuple~str~
    }

    FocusEvaluator <|.. BaseEvaluator
    FocusEvaluator <|.. PenaltyDecorator
    PenaltyDecorator <|-- MovementPenaltyDecorator
    PenaltyDecorator <|-- EyesClosedPenaltyDecorator
    PenaltyDecorator <|-- TiltPenaltyDecorator
    PenaltyDecorator <|-- YawnPenaltyDecorator
    PenaltyDecorator o-- FocusEvaluator : wraps
    PenaltyDecorator ..> Verdict : returns
    ScoreEngine o-- FocusEvaluator : drives
    ScoreEngine --> ScoringConfig : configured by
    ScoreEngine ..> ScoreResult : returns
סדר העטיפה בפועל (evaluator_factory.build_default_focus_evaluator):
BaseEvaluator → MovementPenaltyDecorator → EyesClosedPenaltyDecorator → TiltPenaltyDecorator → YawnPenaltyDecorator
כל שכבה קוראת ל-_wrapped.evaluate(), ואז ממזגת: target_ceiling = min(...), reasons = inner + own, is_violating = OR.

04 / SERVER — API · SECURITY · DOMAIN VALIDATION

שכבת ה-API והאבטחה

ראוטרים דקים, מוגנים ב-Dependency Injection של FastAPI; כל קלט חיצוני עובר ולידציית Pydantic קפדנית לפני שהוא נוגע בלוגיקה עסקית כלשהי.

classDiagram
    class ServerSettings {
        <<Pydantic BaseSettings>>
        +api_key: str
        +dashboard_username: str
        +dashboard_password_hash: str
        +cors_origins: List~str~
        +database_url: str
        +student_timeout_seconds: float
        +rate_limit_requests_per_minute: int
    }

    class require_api_key {
        <<dependency function>>
        "secrets.compare_digest(token, api_key)"
        "→ 401 אם שגוי"
    }

    class require_dashboard_auth {
        <<dependency function>>
        "HTTPBasic + bcrypt verify"
        "→ 401 אם שגוי"
    }

    class RateLimitMiddleware {
        -_hits: Dict~ip, deque~
        -_limit: int
        +dispatch(request, call_next)
    }

    class SecurityHeadersMiddleware {
        +dispatch(request, call_next)
        "X-Frame-Options · X-Content-Type-Options"
    }

    class ScoreSubmission_Schema["ScoreSubmission «Pydantic»"] {
        +student_id: str  ~allowlist regex~
        +focus_score: int  ~0..100~
        +reasons: List~str~  ~max 10~
        +validate student_id()
        +validate reasons()
    }

    class ScoreSubmissionResponse {
        +status: str
        +message: str | None
    }

    class DashboardStudent {
        +score: int
        +last_seen: float
    }

    class ScoresRouter["POST/GET /api/v1/scores"] {
        +submit_score(submission) ScoreSubmissionResponse
        +get_scores() Dict~str,DashboardStudent~
    }

    class DashboardRouter["GET /"] {
        +dashboard() HTMLResponse
    }

    ScoresRouter ..> require_api_key : Depends (POST)
    ScoresRouter ..> require_dashboard_auth : Depends (GET)
    DashboardRouter ..> require_dashboard_auth : Depends
    ScoresRouter ..> ScoreSubmission_Schema : validates body
    ScoresRouter ..> ScoreSubmissionResponse : returns
    ScoresRouter ..> DashboardStudent : returns
    require_api_key ..> ServerSettings : reads api_key
    require_dashboard_auth ..> ServerSettings : reads credentials
הגנה בשכבות: RateLimitMiddleware ו-SecurityHeadersMiddleware עוטפים את כל האפליקציה (ראו תרשים 00), לפני שבקשה בכלל מגיעה לראוטר או ל-Dependency של האימות.
05 / SERVER — DESIGN PATTERN: DECORATOR (שוב) + FACTORY

שכבת הנתונים — Repository מעוטר בהתמדה

אותה שפת עיצוב כמו בלקוח: PersistingStudentRepository עוטף InMemoryStudentRepository ומוסיף עליו כתיבה ל-SQLite, בלי לשנות אף צרכן קיים של IStudentRepository.

classDiagram
    class StudentState {
        <<dataclass, frozen>>
        +score: int
        +reasons: List~str~
        +last_seen: float
    }

    class IStudentRepository {
        <<interface>>
        +upsert(name, score, reasons)*
        +get_active(timeout_seconds) Dict*
    }

    class InMemoryStudentRepository {
        -_students: Dict~str, StudentState~
        -_lock: threading.Lock
        +upsert(name, score, reasons)
        +get_active(timeout_seconds) Dict
    }

    class PersistingStudentRepository {
        -_wrapped: IStudentRepository
        -_session_factory: Callable
        +upsert(name, score, reasons)
        +get_active(timeout_seconds) Dict
        -_persist(name, score, reasons)
    }

    class ScoreLog {
        <<SQLAlchemy ORM Entity>>
        +id: int
        +student_name: str
        +score: int
        +reasons: str
        +created_at: float
    }

    class ScoreService {
        -_repository: IStudentRepository
        -_student_timeout_seconds: float
        +submit_score(submission)
        +get_dashboard_state() Dict~str,DashboardStudent~
    }

    class create_app["server.app.create_app()"] {
        <<Factory>>
        "מרכיבה: InMemory → Persisting → ScoreService"
    }

    IStudentRepository <|.. InMemoryStudentRepository
    IStudentRepository <|.. PersistingStudentRepository
    PersistingStudentRepository o-- IStudentRepository : wraps
    PersistingStudentRepository ..> ScoreLog : INSERT (best-effort)
    InMemoryStudentRepository ..> StudentState : holds
    ScoreService o-- IStudentRepository : depends on interface
    create_app ..> InMemoryStudentRepository : creates
    create_app ..> PersistingStudentRepository : wraps with
    create_app ..> ScoreService : injects into
למה זה חשוב: ה-cache בזיכרון עונה לדשבורד תוך מיקרושניות (ללא פגיעה ב-DB בכל poll), בעוד ה-DB משמש רק כהיסטוריה/ביקורת ברקע — כתיבה שנכשלת ל-DB לא מפילה את הבקשה החיה.
